# Idea Novelty Jury Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an independent novelty jury stage that deduplicates idea cards, records closest prior work, and prevents weak or duplicate ideas from advancing as final portfolio candidates.

**Architecture:** Extend the Divergent mode graph with a `novelty_jury` node between `prior_art_check` and `feasibility_review`. Store jury fields on `idea_card`, persist generated ideas from worker runs, and keep the LLM jury grounded in verified prior-art details generated by the paper verification gate.

**Tech Stack:** Python 3.10, LangGraph-style mode nodes, Pydantic-compatible dict payloads, asyncpg facade, pytest, existing `generate_llm_json` wrapper.

```sql
-- Jury and deduplication fields for generated idea cards.
ALTER TABLE idea_card
    ADD COLUMN IF NOT EXISTS dedup_key TEXT,
    ADD COLUMN IF NOT EXISTS novelty_verdict TEXT NOT NULL DEFAULT 'unclear',
    ADD COLUMN IF NOT EXISTS quality_verdict TEXT NOT NULL DEFAULT 'hold',
    ADD COLUMN IF NOT EXISTS closest_prior_work JSONB NOT NULL DEFAULT '[]'::jsonb,
    ADD COLUMN IF NOT EXISTS strongest_objection TEXT,
    ADD COLUMN IF NOT EXISTS required_validation TEXT[] NOT NULL DEFAULT '{}',
    ADD COLUMN IF NOT EXISTS jury_model TEXT,
    ADD COLUMN IF NOT EXISTS jury_trace_id TEXT,
    ADD COLUMN IF NOT EXISTS jury_status TEXT NOT NULL DEFAULT 'pending',
    ADD COLUMN IF NOT EXISTS prior_art_details JSONB NOT NULL DEFAULT '[]'::jsonb;

UPDATE idea_card
SET dedup_key = lower(regexp_replace(coalesce(title, id::text), '[^a-z0-9]+', '-', 'g'))
WHERE dedup_key IS NULL;

DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'idea_card_novelty_verdict_check'
    ) THEN
        ALTER TABLE idea_card
            ADD CONSTRAINT idea_card_novelty_verdict_check
            CHECK (novelty_verdict IN ('novel', 'incremental', 'duplicate', 'unclear'));
    END IF;

    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'idea_card_quality_verdict_check'
    ) THEN
        ALTER TABLE idea_card
            ADD CONSTRAINT idea_card_quality_verdict_check
            CHECK (quality_verdict IN ('pursue', 'hold', 'reject'));
    END IF;

    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'idea_card_jury_status_check'
    ) THEN
        ALTER TABLE idea_card
            ADD CONSTRAINT idea_card_jury_status_check
            CHECK (jury_status IN ('pending', 'reviewed', 'error'));
    END IF;

    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'idea_card_closest_prior_work_array_check'
    ) THEN
        ALTER TABLE idea_card
            ADD CONSTRAINT idea_card_closest_prior_work_array_check
            CHECK (jsonb_typeof(closest_prior_work) = 'array');
    END IF;

    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'idea_card_prior_art_details_array_check'
    ) THEN
        ALTER TABLE idea_card
            ADD CONSTRAINT idea_card_prior_art_details_array_check
            CHECK (jsonb_typeof(prior_art_details) = 'array');
    END IF;
END $$;

CREATE UNIQUE INDEX IF NOT EXISTS idx_idea_card_run_dedup_key
    ON idea_card(run_id, dedup_key)
    WHERE dedup_key IS NOT NULL;
CREATE INDEX IF NOT EXISTS idx_idea_card_quality_verdict ON idea_card(quality_verdict);
CREATE INDEX IF NOT EXISTS idx_idea_card_novelty_verdict ON idea_card(novelty_verdict);
```

```python
def _idea_dedup_key(card: dict[str, Any]) -> str:
    title = str(card.get("title") or "")
    problem = str(card.get("problem_statement") or "")
    raw_key = f"{title} {problem}".lower()
    key = re.sub(r"[^a-z0-9]+", "-", raw_key).strip("-")
    return key[:140] or "untitled-idea"


def _dedupe_idea_cards(cards: list[dict[str, Any]]) -> list[dict[str, Any]]:
    seen: dict[str, int] = {}
    deduped: list[dict[str, Any]] = []
    for index, card in enumerate(cards):
        next_card = dict(card)
        dedup_key = _idea_dedup_key(next_card)
        next_card["dedup_key"] = dedup_key
        if dedup_key in seen:
            first_index = seen[dedup_key]
            next_card["novelty_verdict"] = "duplicate"
            next_card["quality_verdict"] = "reject"
            next_card["jury_status"] = "reviewed"
            next_card["strongest_objection"] = (
                f"Duplicate of idea {first_index + 1} in this run."
            )
        else:
            seen[dedup_key] = index
            next_card.setdefault("novelty_verdict", "unclear")
            next_card.setdefault("quality_verdict", "hold")
            next_card.setdefault("jury_status", "pending")
        deduped.append(next_card)
    return deduped
```

```python
def _attach_prior_art_details(
    cards: list[dict[str, Any]],
    prior_art_records_by_key: dict[str, list[dict[str, Any]]],
) -> list[dict[str, Any]]:
    updated: list[dict[str, Any]] = []
    for card in cards:
        next_card = dict(card)
        dedup_key = str(next_card.get("dedup_key") or _idea_dedup_key(next_card))
        records = prior_art_records_by_key.get(dedup_key, [])
        verified_records = [
            record for record in records
            if record.get("verification_status") == "verified"
        ]
        next_card["prior_art_details"] = verified_records
        next_card["closest_prior_work"] = [
            {
                "title": record.get("canonical_title") or record.get("input_title"),
                "doi": record.get("canonical_doi"),
                "arxiv_id": record.get("canonical_arxiv_id"),
                "candidate_key": record.get("candidate_key"),
            }
            for record in verified_records[:5]
        ]
        updated.append(next_card)
    return updated
```

```python
_NOVELTY_JURY_SYSTEM = """You are an independent research novelty jury.
Judge each idea against the verified prior-art records supplied with the card.
Do not reward wording changes, agent orchestration language, or broad repackaging.
Use quality_verdict='pursue' only when the mechanism is meaningfully distinct and testable.
```

Add the node:

```python
async def novelty_jury(state: ModeGraphState) -> ModeGraphState:
    """Review generated ideas for novelty and independent objections."""
    reviewable_cards = [
        card for card in state.idea_cards
        if card.get("quality_verdict") != "reject"
    ]
    if not reviewable_cards:
        return state

    payload = {
        "topic": state.topic,
        "ideas": [
            {
                "dedup_key": card.get("dedup_key") or _idea_dedup_key(card),
                "title": card.get("title"),
                "problem_statement": card.get("problem_statement"),
                "mechanism_of_transfer": card.get("mechanism_of_transfer"),
                "expected_benefit": card.get("expected_benefit"),
                "closest_prior_work": card.get("closest_prior_work", []),
                "prior_art_details": card.get("prior_art_details", []),
            }
            for card in reviewable_cards
        ],
    }
    response = await generate_llm_json(
        system_prompt=_NOVELTY_JURY_SYSTEM,
        user_prompt=json.dumps(payload, ensure_ascii=False),
        schema={
            "type": "object",
            "properties": {
                "ideas": {
                    "type": "array",
                    "items": {
                        "type": "object",
                        "properties": {
                            "dedup_key": {"type": "string"},
                            "novelty_verdict": {
                                "type": "string",
                                "enum": ["novel", "incremental", "duplicate", "unclear"],
                            },
                            "quality_verdict": {
                                "type": "string",
                                "enum": ["pursue", "hold", "reject"],
                            },
                            "strongest_objection": {"type": "string"},
                            "required_validation": {
                                "type": "array",
                                "items": {"type": "string"},
                            },
                        },
                        "required": [
                            "dedup_key",
                            "novelty_verdict",
                            "quality_verdict",
                            "strongest_objection",
                            "required_validation",
                        ],
                    },
                }
            },
            "required": ["ideas"],
        },
        temperature=0.1,
    )
    verdicts = {
        item.get("dedup_key"): item
        for item in response.get("ideas", [])
        if item.get("dedup_key")
    }
    updated_cards: list[dict[str, Any]] = []
    for card in state.idea_cards:
        next_card = dict(card)
        verdict = verdicts.get(next_card.get("dedup_key"))
        if verdict:
            next_card["novelty_verdict"] = verdict["novelty_verdict"]
            next_card["quality_verdict"] = verdict["quality_verdict"]
            next_card["strongest_objection"] = verdict["strongest_objection"]
```
